local WaterParts = {}
local OriginalWaterData = {}

local function FindWaterParts()
    WaterParts = {}
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name:lower() == "water" then
            table.insert(WaterParts, obj)
        end
    end
    return #WaterParts
end

local function EnableWalkOnWater()
    local count = FindWaterParts()
    if count == 0 then return false end
    for _, waterPart in ipairs(WaterParts) do
        OriginalWaterData[waterPart] = {CanCollide = waterPart.CanCollide}
        waterPart.CanCollide = true
    end
    return true
end

local function DisableWalkOnWater()
    for waterPart, data in pairs(OriginalWaterData) do
        if waterPart and waterPart.Parent then
            waterPart.CanCollide = data.CanCollide
        end
    end
    OriginalWaterData = {}
end

local PossibleRemotes = {
    "UseSkill", "CastSkill", "Skill", "ActivateSkill",
    "PlaySkill", "SkillActivate", "DoSkill"
}

local function findSkillRemote()
    local events = RS:FindFirstChild("Events") or RS
    for _, name in ipairs(PossibleRemotes) do
        local remote = events:FindFirstChild(name, true)
        if remote and (remote:IsA("RemoteEvent") or remote:IsA("RemoteFunction")) then
            return remote
        end
    end
    return nil
end

local skillRemote = findSkillRemote()

if skillRemote then
    print("✅ Skill Remote found:", skillRemote.Name)
else
    warn("⚠️ Skill remote not found — using VIM fallback")
end

local function useSkill(slot)
    if skillRemote then
        pcall(function()
            skillRemote:FireServer(slot)
            skillRemote:FireServer(slot:lower())
            skillRemote:FireServer({Skill = slot})
            skillRemote:FireServer(1)
        end)
        return
    end
    
    pcall(function()
        VIM:SendKeyEvent(true, Enum.KeyCode[slot], false, game)
        task.wait(0.05)
        VIM:SendKeyEvent(false, Enum.KeyCode[slot], false, game)
    end)
end

local function CreateESP(npc, npcType, color)
    if not npc or not npc:IsA("Model") then return end
    if getgenv().ESPList[npc] then return end
    
    local hrp = npc:FindFirstChild("HumanoidRootPart") or npc:FindFirstChild("PrimaryPart")
    if not hrp then return end
    
    local highlight = Instance.new("Highlight")
    highlight.Name = "NPC_ESP"
    highlight.Adornee = npc
    highlight.FillTransparency = 1

local function GetNPCType(obj)
    if not obj:IsA("Model") then return nil end
    if obj == lp.Character then return nil end
    if Players:GetPlayerFromCharacter(obj) then return nil end
    
    local name = string.lower(obj.Name)
    
    if name:find("taoist") then
        return "Taoist", Color3.fromRGB(255, 165, 0)
    end
    
    if name:find("maoshan") then
        return "Maoshan", Color3.fromRGB(180, 0, 255)
    end
    
    return nil
end

task.spawn(function()
    while task.wait(1) do
        if getgenv().ESP_Taoist or getgenv().ESP_Maoshan then
            for _, obj in ipairs(workspace:GetDescendants()) do
                if obj:IsA("Model") then
                    local npcType, color = GetNPCType(obj)
                    
                    if npcType == "Taoist" and getgenv().ESP_Taoist then
                        CreateESP(obj, npcType, color)
                    elseif npcType == "Maoshan" and getgenv().ESP_Maoshan then
                        CreateESP(obj, npcType, color)
                    end
                end
            end
        end
    end
end)

local function CreatePlayerESP(player)
    if not player or player == lp then return end
    if getgenv().ESPPlayerList[player] then return end
    
    local char = player.Character
    if not char then return end
    
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "Player_ESP"
    billboard.Size = UDim2.fromOffset(200, 30)
    billboard.StudsOffset = Vector3.new(0, 3.5, 0)
    billboard.AlwaysOnTop = true
    billboard.Adornee = hrp
    billboard.Parent = char
    
    local nameLabel = Instance.new("TextLabel")
    nameLabel.Size = UDim2.new(1, 0, 1, 0)
    nameLabel.BackgroundTransparency = 1
    nameLabel.Text = player.Name
    nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    nameLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    nameLabel.TextStrokeTransparency = 0
    nameLabel.Font = Enum.Font.GothamBold
    nameLabel.TextSize = 16
    nameLabel.Parent = billboard
    
    local highlight = Instance.new("Highlight")
    highlight.Name = "Player_Highlight"
    highlight.Adornee = char
    highlight.FillTransparency = 1
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.OutlineTransparency = 0.3
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = char
    
    getgenv().ESPPlayerList[player] = {
        billboard = billboard,
        highlight = highlight
    }
end

local function RemovePlayerESP(player)
    if getgenv().ESPPlayerList[player] then
        local data = getgenv().ESPPlayerList[player]
        if data.billboard then data.billboard:Destroy() end
        if data.highlight then data.highlight:Destroy() end
        getgenv().ESPPlayerList[player] = nil
    end
end

Players.PlayerAdded:Connect(function(player)
    if getgenv().ESP_Player then
        player.CharacterAdded:Connect(function()
            task.wait(1)
            if getgenv().ESP_Player then
                CreatePlayerESP(player)
            end
        end)
    end
end)

Players.PlayerRemoving:Connect(function(player)
    RemovePlayerESP(player)
end)

task.spawn(function()
    while task.wait(1) do
        if getgenv().ESP_Player then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= lp then
                    CreatePlayerESP(player)
                end
            end
        else
            for player, _ in pairs(getgenv().ESPPlayerList) do
                RemovePlayerESP(player)
            end
        end
    end
end)
